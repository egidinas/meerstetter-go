# ============================================================
# PiXtend V2-L CAN configuration — append to /boot/firmware/config.txt
# (Raspberry Pi OS Bookworm) or /boot/config.txt (Bullseye and earlier)
#
# BEFORE APPLYING:
#   1. Set the CAN/analog jumper on the PiXtend V2-L board to the CAN
#      position (see PiXtend V2-L hardware manual, jumper block near the
#      CAN terminal). The board will not enumerate CAN interfaces with the
#      jumper in the analog position.
#   2. Ensure the SD card filesystem is read-write (default on Raspberry
#      Pi OS; overlay-filesystem installs may be read-only).
#
# PREFERRED — single pixtendv2l overlay (sets both MCP2515 at 16 MHz,
# GPIO25/GPIO24 interrupts, SPI CE0/CE1):
# ============================================================

dtparam=spi=on
dtoverlay=pixtendv2l

# Uncomment and remove the dtoverlay=pixtendv2l line above if using this.
